# Panduan Lengkap Optimasi Gambar (Sharp AVIF & WebP) di Windows PowerShell

Dokumen ini dirancang khusus agar kamu dapat mengelola puluhan hingga ratusan foto kamera berukuran besar (**10–20 MB per foto**) tanpa membuat website lambat atau membuat repository GitHub penuh.

---

## 1. Mengapa Tidak Boleh Langsung Memasukkan Foto 10–20 MB ke Website?

1. **Kecepatan Website:** Jika pengunjung membuka galeri berisi 10 foto yang masing-masing berukuran 15 MB, mereka harus mengunduh **150 MB** hanya untuk melihat satu halaman. Di jaringan seluler, gambar akan gagal dimuat atau sangat lambat.
2. **Batas Ukuran GitHub:** GitHub membatasi ukuran repository dan file. Menyimpan ratusan foto mentah di Git akan membuat proses `git push` dan proses build di Vercel menjadi sangat lambat.
3. **Solusi di Proyek Ini:**
   - Foto asli (10–20 MB) disimpan di folder `source-images/` yang **diabaikan oleh Git** (`.gitignore`).
   - Script `npm run images:optimize` menggunakan **Sharp** di laptop Windows kamu untuk membuat versi ringan berformat **AVIF** dan **WebP** dalam beberapa ukuran layar (`480w`, `768w`, `1200w`, `1600w`, `2048w`), sekaligus menghapus data lokasi GPS dari EXIF demi privasi.
   - Hasil ukurannya turun drastis dari **15 MB menjadi sekitar 120 KB – 350 KB** tanpa terlihat pecah!

---

## 2. Mengenal Perbedaan Format Gambar

- **Original JPG / PNG:** File mentah dari kamera atau software desain (misal Canva/Figma). Ukurannya sangat besar (10–20 MB). Hanya disimpan di laptop lokal (`source-images/`).
- **AVIF (`.avif`):** Format gambar generasi terbaru dengan kompresi paling efisien dan detail warna yang sangat halus. Browser modern (Chrome, Edge, Firefox, Safari terbaru) akan otomatis memilih file ini terlebih dahulu.
- **WebP (`.webp`):** Format gambar modern yang didukung hampir seluruh browser sebagai *fallback* (cadangan) apabila browser pengunjung belum mendukung AVIF.
- **Responsive Image Variants (`480w` s.d. `2048w`):**
  - Layar HP hanya perlu mengunduh gambar lebar `480px` atau `768px`.
  - Layar monitor laptop/PC atau saat membuka *Lightbox* layar penuh akan otomatis mengambil ukuran `1600px` atau `2048px`.
- **Thumbnail vs Full-size Image:** Kamu tidak perlu memotong gambar *thumbnail* secara manual di Photoshop/Canva. Satu foto original otomatis dibuatkan ukuran kecil (untuk *card/grid*) dan ukuran besar (untuk *Lightbox*).

import fs from "node:fs/promises";
import path from "node:path";
import sharp from "sharp";
sharp.cache({ files: 50, memory: 512 }); // Nonaktifkan cache internal Sharp untuk menghemat memori

const ROOT_DIR = process.cwd();
const SOURCE_DIR = path.join(ROOT_DIR, "source-images");
const OUTPUT_DIR = path.join(ROOT_DIR, "public", "images", "optimized");
const MANIFEST_PATH = path.join(ROOT_DIR, "src", "generated", "image-manifest.json");

const TARGET_WIDTHS = [480, 768, 1200, 1600, 2048];
const SUPPORTED_EXTENSIONS = new Set([".jpg", ".jpeg", ".png", ".webp", ".tiff"]);

/**
 * Konfigurasi preset kompresi berdasarkan kategori folder.
 * - Photography: mengutamakan rentang dinamis, ketajaman tekstur alam/kegiatan, dan kompresi seimbang.
 * - Social Media & Projects: mengutamakan ketajaman tipografi/garis desain poster agar teks tidak blur.
 */
const COMPRESSION_PRESETS = {
  photography: {
    avif: { quality: 68, effort: 4 },
    webp: { quality: 82, effort: 4 },
  },
  "social-media": {
    avif: { quality: 80, effort: 4 },
    webp: { quality: 90, effort: 4, nearLossless: false },
  },
  default: {
    avif: { quality: 74, effort: 4 },
    webp: { quality: 85, effort: 4 },
  },
};

async function getFilesRecursively(dir) {
  let results = [];
  try {
    const entries = await fs.readdir(dir, { withFileTypes: true });
    for (const entry of entries) {
      const fullPath = path.join(dir, entry.name);
      if (entry.isDirectory()) {
        results = results.concat(await getFilesRecursively(fullPath));
      } else if (entry.isFile()) {
        const ext = path.extname(entry.name).toLowerCase();
        if (SUPPORTED_EXTENSIONS.has(ext)) {
          results.push(fullPath);
        }
      }
    }
  } catch (err) {
    if (err.code !== "ENOENT") {
      console.error(`Gagal membaca direktori ${dir}:`, err.message);
    }
  }
  return results;
}

function formatBytes(bytes) {
  if (bytes === 0) return "0 B";
  const k = 1024;
  const sizes = ["B", "KB", "MB", "GB"];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(2))} ${sizes[i]}`;
}

async function runOptimization() {
  console.log("============================================================");
  console.log("  SHARP IMAGE OPTIMIZATION PIPELINE — AURELL PORTFOLIO");
  console.log("============================================================");

  await fs.mkdir(OUTPUT_DIR, { recursive: true });
  await fs.mkdir(path.dirname(MANIFEST_PATH), { recursive: true });

  const files = await getFilesRecursively(SOURCE_DIR);

  if (files.length === 0) {
    console.log("ℹ Tidak ditemukan gambar di folder source-images/.");
    try {
      await fs.access(MANIFEST_PATH);
    } catch {
      await fs.writeFile(MANIFEST_PATH, JSON.stringify({}, null, 2), "utf-8");
    }
    console.log("✔ Manifest kosong siap digunakan.\n");
    return;
  }

  const manifest = {};
  let totalOriginalBytes = 0;
  let totalLargestOptimizedBytes = 0;

  for (const filePath of files) {
    const relFromSource = path.relative(SOURCE_DIR, filePath);
    const normalizedRel = relFromSource.split(path.sep).join("/");
    const parsed = path.parse(normalizedRel);

    // Key unik tanpa ekstensi, misal: "photography/merbabu-01"
    const imageKey = parsed.dir ? `${parsed.dir}/${parsed.name}` : parsed.name;
    const category = parsed.dir.split("/")[0] || "default";
    const preset = COMPRESSION_PRESETS[category] || COMPRESSION_PRESETS.default;

    const stat = await fs.stat(filePath);
    totalOriginalBytes += stat.size;

    // Baca gambar dan auto-orient berdasarkan EXIF
    const image = sharp(filePath, {limitInputPixels: false}).rotate();
    const metadata = await image.metadata();

    // Setelah .rotate(), jika orientasi EXIF >= 5 (90/270 deg), tukar dimensi efektif
    const isSwapped = metadata.orientation && metadata.orientation >= 5;
    const origWidth = isSwapped ? metadata.height : metadata.width;
    const origHeight = isSwapped ? metadata.width : metadata.height;

    if (!origWidth || !origHeight) {
      console.warn(`⚠ Melewati ${normalizedRel}: dimensi gambar tidak terbaca.`);
      continue;
    }

    const aspectRatio = Number((origWidth / origHeight).toFixed(4));

    // Tentukan daftar ukuran yang tidak melebihi ukuran asli (tanpa upscaling)
    const applicableWidths = TARGET_WIDTHS.filter((w) => w <= origWidth);
    if (applicableWidths.length === 0 || applicableWidths[applicableWidths.length - 1] !== origWidth) {
      if (origWidth < TARGET_WIDTHS[0]) {
        applicableWidths.push(origWidth);
      } else if (origWidth <= 2048 && !applicableWidths.includes(origWidth)) {
        applicableWidths.push(origWidth);
      }
    }

    const targetOutDir = path.join(OUTPUT_DIR, parsed.dir);
    await fs.mkdir(targetOutDir, { recursive: true });

    const avifVariants = [];
    const webpVariants = [];
    let largestAvifSize = 0;

    for (const width of applicableWidths) {
      const height = Math.round(width / aspectRatio);
      const avifFilename = `${parsed.name}-${width}w.avif`;
      const webpFilename = `${parsed.name}-${width}w.webp`;

      const avifFullOut = path.join(targetOutDir, avifFilename);
      const webpFullOut = path.join(targetOutDir, webpFilename);

      const publicBasePath = `/images/optimized/${parsed.dir ? parsed.dir + "/" : ""}`;

      // Pipeline menghapus metadata GPS/EXIF pribadi secara default pada Sharp
      const pipeline = sharp(filePath, {limitInputPixels: false})
        .rotate()
        .resize({
          width,
          withoutEnlargement: true,
          fit: "inside",
        });

      const avifInfo = await pipeline.clone().avif(preset.avif).toFile(avifFullOut);
      const webpInfo = await pipeline.clone().webp(preset.webp).toFile(webpFullOut);

      avifVariants.push({
        width: avifInfo.width,
        height: avifInfo.height,
        path: `${publicBasePath}${avifFilename}`,
        sizeBytes: avifInfo.size,
      });

      webpVariants.push({
        width: webpInfo.width,
        height: webpInfo.height,
        path: `${publicBasePath}${webpFilename}`,
        sizeBytes: webpInfo.size,
      });

      if (avifInfo.size > largestAvifSize) {
        largestAvifSize = avifInfo.size;
      }
    }

    totalLargestOptimizedBytes += largestAvifSize;

    const highestVariant = webpVariants[webpVariants.length - 1];

    manifest[imageKey] = {
      key: imageKey,
      originalFilename: parsed.base,
      category,
      originalWidth: origWidth,
      originalHeight: origHeight,
      outputWidth: highestVariant.width,
      outputHeight: highestVariant.height,
      aspectRatio,
      originalSizeBytes: stat.size,
      avif: avifVariants,
      webp: webpVariants,
      fallbackSrc: highestVariant.path,
    };

    console.log(
      `✔ [${imageKey}] ${formatBytes(stat.size)} -> AVIF max ${formatBytes(largestAvifSize)} (${applicableWidths.join("w, ")}w)`
    );
  }

  await fs.writeFile(MANIFEST_PATH, JSON.stringify(manifest, null, 2), "utf-8");

  const savedPercent =
    totalOriginalBytes > 0
      ? (((totalOriginalBytes - totalLargestOptimizedBytes) / totalOriginalBytes) * 100).toFixed(1)
      : "0";

  console.log("------------------------------------------------------------");
  console.log(`Total Gambar Diproses : ${Object.keys(manifest).length}`);
  console.log(`Total Ukuran Original : ${formatBytes(totalOriginalBytes)}`);
  console.log(`Total Ukuran Optimal  : ${formatBytes(totalLargestOptimizedBytes)} (Hemat ~${savedPercent}%)`);
  console.log(`Manifest Tersimpan di : src/generated/image-manifest.json`);
  console.log("============================================================\n");
}

runOptimization().catch((err) => {
  console.error("❌ Gagal menjalankan optimasi gambar:", err);
  process.exit(1);
});

            key={cat}

            className={`px-3.5 py-1.5 rounded-lg text-xs font-mono border transition-colors ${

            }`}
          >
            {cat}
          </button>
        ))}
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {filteredProjects.map((project) => (
          <ProjectCard key={project.slug} project={project} />
        ))}
      </div>
    </div>
  );
}

"use client";

import React, { useEffect, useCallback } from "react";
import { X, ChevronLeft, ChevronRight } from "lucide-react";
import { ResponsiveImage } from "@/components/ui/ResponsiveImage";

export interface LightboxSlide {
  imageKey: string;
  alt: string;
  caption?: string;
  meta?: string;
}

interface LightboxProps {
  slides: LightboxSlide[];
  activeIndex: number | null;
  onClose: () => void;
  onChangeIndex: (newIndex: number) => void;
}

export function Lightbox({
  slides,
  activeIndex,
  onClose,
  onChangeIndex,
}: LightboxProps) {
  const isOpen = activeIndex !== null && activeIndex >= 0 && activeIndex < slides.length;

  const handlePrev = useCallback(() => {
    if (activeIndex === null) return;
    onChangeIndex((activeIndex - 1 + slides.length) % slides.length);
  }, [activeIndex, slides.length, onChangeIndex]);

  const handleNext = useCallback(() => {
    if (activeIndex === null) return;
    onChangeIndex((activeIndex + 1) % slides.length);
  }, [activeIndex, slides.length, onChangeIndex]);

  useEffect(() => {
    if (!isOpen) return;

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
      if (e.key === "ArrowLeft") handlePrev();
      if (e.key === "ArrowRight") handleNext();
    };

    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [isOpen, onClose, handlePrev, handleNext]);

  if (!isOpen || activeIndex === null) return null;

  const current = slides[activeIndex];

  return (
    <div
      role="dialog"
      aria-modal="true"
      aria-label="Image Lightbox"
      className="fixed inset-0 z-50 bg-black/95 backdrop-blur-sm flex flex-col justify-between p-4 sm:p-6"
    >
      <div className="flex items-center justify-between text-xs font-mono text-zinc-400">
        <span>
          {activeIndex + 1} / {slides.length}
        </span>
        <button
          type="button"
          onClick={onClose}
          aria-label="Close Lightbox"
          className="p-2 rounded-lg bg-zinc-900 border border-zinc-800 text-zinc-200 hover:border-lime-400 transition-colors"
        >
          <X className="w-5 h-5" />
        </button>
      </div>

      <div className="relative flex-1 flex items-center justify-center my-4 overflow-hidden">
        {slides.length > 1 && (
          <button
            type="button"
            onClick={handlePrev}
            aria-label="Previous Image"
            className="absolute left-2 sm:left-6 z-10 p-2.5 rounded-full bg-zinc-900/90 border border-zinc-700 text-zinc-100 hover:border-lime-400 transition-colors"
          >
            <ChevronLeft className="w-5 h-5" />
          </button>
        )}

        <div className="max-h-[78vh] max-w-5xl w-full h-full flex items-center justify-center">
          <ResponsiveImage
            imageKey={current.imageKey}
            alt={current.alt}
            sizes="100vw"
            objectFit="contain"
            className="w-full h-full flex items-center justify-center"
            imgClassName="max-h-[78vh] w-auto h-auto mx-auto object-contain"
          />
        </div>

        {slides.length > 1 && (
          <button
            type="button"
            onClick={handleNext}
            aria-label="Next Image"
            className="absolute right-2 sm:right-6 z-10 p-2.5 rounded-full bg-zinc-900/90 border border-zinc-700 text-zinc-100 hover:border-lime-400 transition-colors"
          >
            <ChevronRight className="w-5 h-5" />
          </button>
        )}
      </div>

      {(current.caption || current.meta) && (
        <div className="max-w-2xl mx-auto text-center space-y-1 pb-2">
          {current.caption && (
            <p className="text-sm text-zinc-200">{current.caption}</p>
          )}
          {current.meta && (
            <p className="text-xs font-mono text-zinc-500">{current.meta}</p>
          )}
        </div>
      )}
    </div>
  );
}

import React from "react";
import manifestData from "@/generated/image-manifest.json";
import { ManifestEntry } from "@/types/portfolio";
import { ImageOff } from "lucide-react";

const manifest = manifestData as Record<string, ManifestEntry>;

interface ResponsiveImageProps {
  imageKey?: string | null;
  alt: string;
  sizes?: string;
  className?: string;
  imgClassName?: string;
  priority?: boolean;
  objectFit?: "cover" | "contain";
}

export function ResponsiveImage({
  imageKey,
  alt,
  sizes = "(max-width: 640px) 100vw, (max-width: 1024px) 50vw, 33vw",
  className = "",
  imgClassName = "",
  priority = false,
  objectFit = "cover",
}: ResponsiveImageProps) {
  if (!imageKey) {
    return (
      <div
        className={`flex flex-col items-center justify-center bg-zinc-900/70 border border-zinc-800/80 text-zinc-500 p-6 text-center ${className}`}
      >
        <ImageOff className="w-6 h-6 mb-2 text-zinc-600" />
        <span className="text-xs font-mono uppercase tracking-wider text-zinc-500">
          Visual Asset Pending
        </span>
      </div>
    );
  }

  const entry = manifest[imageKey];

  if (!entry) {
    if (process.env.NODE_ENV === "development") {
      console.warn(
        `[ResponsiveImage] Image key "${imageKey}" tidak ditemukan di src/generated/image-manifest.json. Pastikan gambar sudah ada di source-images/ dan jalankan "npm run images:optimize".`
      );
    }

    return (
      <div
        className={`flex flex-col items-center justify-center bg-zinc-900/70 border border-dashed border-zinc-800 text-zinc-400 p-6 text-center ${className}`}
      >
        <span className="text-xs font-mono text-amber-400/90 mb-1">
          Unoptimized / Missing Key
        </span>
        <code className="text-[11px] text-zinc-500 break-all">{imageKey}</code>
      </div>
    );
  }

  const avifSrcSet = entry.avif.map((v) => `${v.path} ${v.width}w`).join(", ");
  const webpSrcSet = entry.webp.map((v) => `${v.path} ${v.width}w`).join(", ");

  return (
    <picture className={`block overflow-hidden ${className}`}>
      {avifSrcSet && (
        <source type="image/avif" srcSet={avifSrcSet} sizes={sizes} />
      )}
      {webpSrcSet && (
        <source type="image/webp" srcSet={webpSrcSet} sizes={sizes} />
      )}
      <img
        src={entry.fallbackSrc}
        alt={alt}
        width={entry.outputWidth}
        height={entry.outputHeight}
        loading={priority ? "eager" : "lazy"}
        decoding={priority ? "sync" : "async"}
        fetchPriority={priority ? "high" : "auto"}
        className={`w-full h-full transition-transform duration-500 ${
          objectFit === "contain" ? "object-contain" : "object-cover"
        } ${imgClassName}`}
      />
    </picture>
  );
}
